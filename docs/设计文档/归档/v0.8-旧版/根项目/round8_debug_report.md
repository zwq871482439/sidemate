    if len(arr) <= 1:

def quick_sort(arr):
    """
    对数组进行快速排序
    
    参数:
        arr (list): 需要排序的列表
        
    返回值: 
        list：已按升序排列的新列表
    """

    if len(arr) <= 1:
        return arr

    pivot = arr[0]

    left = [x for x in arr if x < pivot]   # 小于区域

    return quick_sort(left) + [pivot] + quick_sort(right)
